# Examples

Recipes for common tasks. They assume `import static ch.lxrin.ql.QL.*;`,
`import static com.example.db.Tables.*;` and tables generated from:

```sql
CREATE TABLE customer (id uuid PRIMARY KEY, name text NOT NULL, email text NOT NULL UNIQUE, country text NOT NULL,
                       tags text[] NOT NULL DEFAULT '{}', profile jsonb NOT NULL DEFAULT '{}',
                       created_at timestamptz NOT NULL DEFAULT now(), deleted_at timestamptz);
CREATE TABLE orders   (id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, customer_id uuid NOT NULL REFERENCES customer,
                       total numeric(12, 2) NOT NULL, status text NOT NULL, ordered_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE job      (id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, status text NOT NULL,
                       started_at timestamptz, created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE booking  (id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, room_id bigint NOT NULL,
                       period tstzrange NOT NULL, status text NOT NULL, kind text NOT NULL);
```

## 1. Search form with optional filters

Only the filters that are set end up in the SQL, and every value is bound. The
sort parameter is mapped through a whitelist.

```java
record CustomerSummary(UUID id, String name, String country) {}

static final Map<String, Field<?>> SORTABLE = Map.of("name", CUSTOMER.NAME, "created", CUSTOMER.CREATED_AT);

public List<CustomerSummary> search(CustomerFilter f, String sort) {
    Condition filter = Conditions.builder(CUSTOMER.DELETED_AT.isNull())
            .addIfPresent(f.name(), CUSTOMER.NAME::containsIgnoreCase)       // Optional<String>
            .addIfPresent(f.country(), CUSTOMER.COUNTRY::eq)
            .addIfPresent(f.tags(), tags -> arrayOverlaps(CUSTOMER.TAGS, tags.toArray(String[]::new)))
            .addIfPresent(f.createdAfter(), CUSTOMER.CREATED_AT::ge)
            .build();
    return createContribution(CustomerSummary.class, CUSTOMER, (c, b) -> c
                    .select(CUSTOMER.ID, CUSTOMER.NAME, CUSTOMER.COUNTRY)
                    .where(filter)
                    .orderBy(Sorts.from(sort, SORTABLE, CUSTOMER.NAME.asc()))  // "created,desc"; unknown keys → InvalidSortException
                    .limit(200))
            .fetch();
}
```

The same with the `select` style and `whereIf`:

```java
selectFrom(CUSTOMER)
        .whereIf(f.name().isPresent(), () -> CUSTOMER.NAME.containsIgnoreCase(f.name().get()))
        .where(CUSTOMER.COUNTRY.eqIfPresent(f.country()))
        .orderBy(Sorts.from(sort, SORTABLE, CUSTOMER.NAME.asc()))
        .limit(200)
        .fetch();
```

## 2. Infinite scrolling with keyset pagination

```java
public Page<CustomerRow> page(String cursor) {
    return selectFrom(CUSTOMER)
            .where(CUSTOMER.COUNTRY.eq("CH"))
            .orderBy(CUSTOMER.CREATED_AT.desc(), CUSTOMER.ID.desc())
            .seekAfterCursor(cursor)
            .limit(50)
            .fetchPage();       // page.items(), page.nextCursor().map(Cursor::encode)
}
```

For page numbers, use `page(index, size)` together with `fetchCount()` on the same
query.

## 3. Revenue per month with a running total

```java
record MonthlyRevenue(Instant month, BigDecimal revenue, BigDecimal runningTotal, long orders) {}

TemporalField<Instant> month = ORDERS.ORDERED_AT.truncate(DatePart.MONTH, ZoneId.of("Europe/Zurich"));

List<MonthlyRevenue> report = select(month.as("month"),
            sum(ORDERS.TOTAL),
            sum(sum(ORDERS.TOTAL)).over(orderBy(month.asc())),
            count())
        .from(ORDERS)
        .where(ORDERS.STATUS.eq("PAID"), ORDERS.ORDERED_AT.ge(Instant.now().minus(Duration.ofDays(365))))
        .groupBy(month)
        .orderBy(month.asc())
        .fetch(MonthlyRevenue::new);
```

## 4. Top N per group

The three largest orders per customer:

```java
Field<Long> rn = rowNumber().over(partitionBy(ORDERS.CUSTOMER_ID).orderBy(ORDERS.TOTAL.desc())).as("rn");
DerivedTable ranked = select(ORDERS.CUSTOMER_ID, ORDERS.ID, ORDERS.TOTAL, rn).from(ORDERS).asTable("r");

List<Row3<UUID, Long, BigDecimal>> top3 = select(ranked.field(ORDERS.CUSTOMER_ID), ranked.field(ORDERS.ID), ranked.field(ORDERS.TOTAL))
        .from(ranked)
        .where(ranked.field(rn).le(3L))
        .fetch();
```

For just the latest row per group, `DISTINCT ON` is shorter:

```java
select(ORDERS.CUSTOMER_ID, ORDERS.ID, ORDERS.ORDERED_AT)
    .distinctOn(ORDERS.CUSTOMER_ID)
    .from(ORDERS)
    .orderBy(ORDERS.CUSTOMER_ID.asc(), ORDERS.ORDERED_AT.desc())
    .fetch();
```

## 5. Latest related row with LATERAL

```java
DerivedTable latest = select(ORDERS.TOTAL, ORDERS.ORDERED_AT).from(ORDERS)
        .where(ORDERS.CUSTOMER_ID.eq(CUSTOMER.ID))
        .orderBy(ORDERS.ORDERED_AT.desc())
        .limit(1)
        .asTable("lo");

select(CUSTOMER.NAME, latest.field(ORDERS.TOTAL), latest.field(ORDERS.ORDERED_AT))
    .from(CUSTOMER)
    .leftJoin(lateral(latest)).onTrue()
    .fetch();
```

## 6. Upsert

```java
insertInto(CUSTOMER)
    .set(CUSTOMER.ID, customers.createKey())
    .set(CUSTOMER.EMAIL, dto.email())
    .set(CUSTOMER.NAME, dto.name())
```

## 7. Job queue with SKIP LOCKED

Several workers take jobs concurrently without blocking each other:

```java
List<Long> claimed = ctx.transaction(() -> update(JOB)
        .set(JOB.STATUS, "RUNNING")
        .set(JOB.STARTED_AT, now())
        .where(JOB.ID.in(select(JOB.ID).from(JOB)
                .where(JOB.STATUS.eq("QUEUED"))
                .orderBy(JOB.CREATED_AT.asc())
                .limit(10)
                .forUpdate().skipLocked()))
        .returning(JOB.ID)
        .fetch());
```

## 8. JSONB

```java
select(CUSTOMER.NAME, CUSTOMER.PROFILE.getText("city"), CUSTOMER.PROFILE.pathText("address", "zip"))
    .from(CUSTOMER)
    .where(CUSTOMER.PROFILE.containsJson("{\"newsletter\": true}"))     // index-friendly @>
    .fetch();

update(CUSTOMER)
    .set(CUSTOMER.PROFILE, jsonbSet(CUSTOMER.PROFILE, new String[]{"newsletter"}, toJsonb(param(false)), true))
    .where(CUSTOMER.ID.eq(id))
    .execute();

String json = select(jsonbAgg(jsonbBuildObject(pair("id", ORDERS.ID), pair("total", ORDERS.TOTAL))).orderBy(ORDERS.ORDERED_AT.asc()))
    .from(ORDERS).where(ORDERS.CUSTOMER_ID.eq(id)).fetchOne();

selectFrom(CUSTOMER).where(CUSTOMER.PROFILE.pathExists("$.orders[*] ? (@.total > 100)"));
```

With a forced type such as `SqlTypes.jsonb(Profile.class)`, the column reads and
writes `Profile` records directly (through the context's `JsonCodec`).

## 10. Full-text search with ranking

```java
Field<String> doc = toTsvector("english", concatWs(" ", CUSTOMER.NAME, CUSTOMER.PROFILE.getText("bio")));
Field<String> query = websearchToTsquery("english", userInput);        // user input is bound

select(CUSTOMER.ID, CUSTOMER.NAME, tsRank(doc, query), tsHeadline("english", CUSTOMER.PROFILE.getText("bio"), query))

    .fetch();
```

## 11. Tree with a recursive CTE

```java
Cte tree = recursiveCte("tree");
NumberColumn<Long> id = tree.declareNumber("id", SqlTypes.INT8);
StringColumn name = tree.declareString("name", SqlTypes.TEXT);
NumberColumn<Integer> depth = tree.declareNumber("depth", SqlTypes.INT4);
tree.as(select(CATEGORY.ID, CATEGORY.NAME, inline(0)).from(CATEGORY).where(CATEGORY.PARENT_ID.isNull())
        .unionAll(select(CATEGORY.ID, CATEGORY.NAME, depth.plus(1)).from(CATEGORY).join(tree).on(tree.field(CATEGORY.ID).eq(CATEGORY.PARENT_ID))));

List<Row3<Long, String, Integer>> rows = select(id, name, depth).withRecursive(tree).from(tree).orderBy(depth.asc(), name.asc()).fetch();
```

## 12. Archiving in one statement

```java
Cte moved = cte("moved", deleteFrom(ORDERS).where(ORDERS.ORDERED_AT.lt(Instant.now().minus(Duration.ofDays(730))))
        .returning(ORDERS.ID, ORDERS.CUSTOMER_ID, ORDERS.TOTAL));
insertInto(ORDERS_ARCHIVE).columns(ORDERS_ARCHIVE.ID, ORDERS_ARCHIVE.CUSTOMER_ID, ORDERS_ARCHIVE.TOTAL)
    .select(select(moved.field(ORDERS.ID), moved.field(ORDERS.CUSTOMER_ID), moved.field(ORDERS.TOTAL)).from(moved))
    .with(moved)
    .execute();
```

## 14. Reusable query fragments

Conditions and fields are plain objects, so you can build them once and reuse them:

```java
final class CustomerQueries {
    static Condition isActive(CustomerTable c) {
        return c.DELETED_AT.isNull().and(c.COUNTRY.ne("XX"));
    }

    static Field<String> displayName(CustomerTable c) {
        return c.NAME.trim().nullIf("").coalesce(c.EMAIL);
    }
}

select(CustomerQueries.displayName(CUSTOMER).as("name")).from(CUSTOMER).where(CustomerQueries.isActive(CUSTOMER)).fetch();
```

## 15. PATCH endpoint

Only the fields that the client sent are written. The version check makes concurrent
changes visible.

```java
record CustomerPatch(Optional<String> name, Optional<String> email, Optional<String> country) {}

long updated = createUpdate(CUSTOMER, (c, b) -> c
        .setIfPresent(CUSTOMER.NAME, patch.name())
        .setIfPresent(CUSTOMER.EMAIL, patch.email())
        .setIfPresent(CUSTOMER.COUNTRY, patch.country())
        .where(CUSTOMER.ID.eq(id), CUSTOMER.DELETED_AT.isNull()))
    .execute();
```
